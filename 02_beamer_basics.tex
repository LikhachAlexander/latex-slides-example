% ---------- Слайд 5: Минимальный пример ----------
\begin{frame}[fragile,label=minimal]{Минимальный рабочий пример}
  % [T] означает выравнивание по верхнему краю (Top)
  \begin{columns}[T,onlytextwidth]
    % --- ЛЕВАЯ КОЛОНКА: КОД ---
    \column{0.52\textwidth}
      \textbf{Исходный код:}
      \vspace{0.2em}
      \inputminted[fontsize=\scriptsize, frame=single, breaklines=true]{latex}{examples/minimal.tex}

    % --- ПРАВАЯ КОЛОНКА: РЕЗУЛЬТАТ ---
    \column{0.44\textwidth}
      \textbf{Результат:}
      \vspace{0.2em}
      \setbeamercolor{normal text}{fg=black, bg=white}
      \begin{beamercolorbox}[wd=\linewidth, sep=4pt, rounded=true, shadow=true]{normal text}
        \small
        \begin{center}
          \textbf{Заголовок слайда}
        \end{center}
        \vspace{0.5em}
        Текст слайда.
      \end{beamercolorbox}
  \end{columns}

  \vspace{0.5em}
  \centering
  \emph{Команда \cmd{usetheme} в преамбуле задает единый стиль всех слайдов.}
  \note{Обратите внимание: \cmd{begin\{frame\}} создает слайд, а его заголовок
  передается в фигурных скобках.}
\end{frame}

% ---------- Слайд 6: Структура проекта ----------
\begin{frame}[fragile,label=structure]{Структура проекта}
  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
      \textbf{Файлы проекта:}
      \vspace{0.2em}
      \inputminted[fontsize=\scriptsize, frame=single, breaklines=true]{text}{examples/tree.txt}

    \column{0.44\textwidth}
      \textbf{Рекомендации:}
      \vspace{0.2em}
      % Используем стандартный itemize без блока, чтобы выровнять с рамкой кода
      \begin{itemize}
        \item \cmd{input\{...\}} подключает содержимое;
        \item Изображения размещаются в \texttt{images/};
        \item Разделы --- в отдельных файлах;
        \item Главный задает структуру.
      \end{itemize}
  \end{columns}
  \note{При значительном объёме проекта единственный файл
  затрудняет навигацию и поиск нужного раздела.}
\end{frame}
